<script>
document.addEventListener("DOMContentLoaded", function () {
  let danmakuContainer = document.querySelector(".danmaku-screen");
  if (!danmakuContainer) {
    danmakuContainer = document.createElement("div");
    danmakuContainer.className = "danmaku-screen";
    document.body.appendChild(danmakuContainer);
  }

  let isDanmakuEnabled = true;

  const toggleBtn = document.getElementById("danmaku-toggle-btn");
  if (toggleBtn) {
    toggleBtn.addEventListener("click", function () {
      isDanmakuEnabled = !isDanmakuEnabled;

      if (isDanmakuEnabled) {
        toggleBtn.classList.remove("off");
        toggleBtn.innerHTML = "특수효과를 표시";
      } else {
        toggleBtn.classList.add("off");
        toggleBtn.innerHTML = "특수효과를 비표시";
        danmakuContainer.innerHTML = "";
      }
    });
  }

  function launchDanmaku(emojisText, count = 25) {
    if (!isDanmakuEnabled) return;

    const emojiList = emojisText.split(",").map(s => s.trim());

    for (let i = 0; i < count; i++) {
      setTimeout(() => {
        if (!isDanmakuEnabled) return;
        const item = document.createElement("div");
        item.className = "danmaku-item";
        
        const randomEmoji = emojiList[Math.floor(Math.random() * emojiList.length)];

        const repeatCount = Math.floor(Math.random() * 3) + 1;
        item.textContent = Array(repeatCount).fill(randomEmoji).join("　");

        const topPos = Math.random() * 90 + 5;
        item.style.top = topPos + "vh";

        const fontSize = 48;
        item.style.fontSize = fontSize + "px";

        const duration = 7;
        item.style.animationDuration = duration + "s";

        danmakuContainer.appendChild(item);

        item.addEventListener("animationend", () => {
          item.remove();
        });
      }, i * 80 + Math.random() * 200);
    }
  }

  const triggers = document.querySelectorAll(".danmaku-trigger");
  
  const COOLDOWN_TIME = 14000;

  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {

      if (entry.isIntersecting && !entry.target.dataset.cooldown) {
        
        entry.target.dataset.cooldown = "true";

        const emojiData = entry.target.getAttribute("data-danmaku");
        if (emojiData) {
          launchDanmaku(emojiData, 24);
        }

        setTimeout(() => {
          delete entry.target.dataset.cooldown;
        }, COOLDOWN_TIME);
      }
    });
  }, {
    rootMargin: "-49% 0px -51% 0px",
    threshold: 0
  });

  triggers.forEach(target => observer.observe(target));
});
</script>
