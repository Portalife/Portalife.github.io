postlist.html 기존 코드

{% comment %}
to use:
{% include postlist.html %}
or
{% include postlist.html category="tech" limit=6 %}

우선순위
1 include로 넘긴 category
2 page.categories의 첫 번째
3 URL에서 /categories/다음 세그먼트
{% endcomment %}

{% assign limit = include.limit | default: 20 %}

{% assign cat = include.category %}
{% if cat == nil %}
  {% if page.categories and page.categories.size > 0 %}
    {% assign cat = page.categories[0] %}
  {% endif %}
{% endif %}

{% if cat == nil %}
  {% assign parts = page.url | split: '/' %}
  {% for p in parts %}
    {% if p == "categories" %}
      {% assign idx = forloop.index %}
      {% assign cat = parts[idx] %}
      {% break %}
    {% endif %}
  {% endfor %}
{% endif %}

{% if cat %}
{% assign raw_posts = site.categories[cat] %}
{% if raw_posts and raw_posts.size > 0 %}
{% assign posts = raw_posts | sort: "order" | reverse | sort: "date" | reverse | slice: 0, limit %}
<div class="postlist">
  <div class="postlist-title">
    {% t sidebar.{{ cat }} %}{% t layout.at %}{{ posts.size }}{% t layout.number %}{% t layout.exist %}
  </div><br>
  <div class="postlist-table">
    {% for post in posts %}
    <div class="postlist-row">
      <div class="postlist-title-cell">
        <a href="{{ post.url | prepend: site.baseurl }}">{{ post.title }}</a>
      </div>
      <div class="postlist-date-cell">
        {{ post.date | date: "%Y.%m.%d" }}
      </div>
    </div>
    {% endfor %}
  </div>
</div>
{% else %}
<div class="postlist">
  <div class="postlist-empty">{% t layout.empty %}</div>
</div>
{% endif %}
